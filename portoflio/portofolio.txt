    <script>
        function ouvrirGalerie(nomGalerie, boutonClique) {
            // 1. On cache toutes les galeries
            let galeries = document.querySelectorAll('.galerie');
            galeries.forEach(g => g.classList.remove('active'));

            // 2. On enlève la couleur active de tous les boutons
            let boutons = document.querySelectorAll('.nav-btn');
            boutons.forEach(b => b.classList.remove('active'));

            // 3. On affiche la galerie choisie et on colore le bouton cliqué
            document.getElementById(nomGalerie).classList.add('active');
            boutonClique.classList.add('active');
        }

        function changerTheme() {
            document.body.classList.toggle('dark-mode');
        }
    </script>
